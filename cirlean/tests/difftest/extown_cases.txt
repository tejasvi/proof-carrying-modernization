# extown.c: a program's own functions named like libevent's and POSIX's (threads, scheduling,
# strerror_r: names C does not reserve, C11 7.1.3), or glibc's __errno_location, run their bodies:
# the library models attach only to declarations without a body. Each case is labelled on the line
# above it (before the change, the models replaced the bodies).
# expect MATCH: the program's evbuffer_new (77; before: UB, the model's buffer read as the program's struct)
extown x_own_evbuffer_new 0
extown x_own_evbuffer_new 5
# expect MATCH: the program's pthread_self (42; before: the model's thread id 1)
extown x_own_pthread_self 0
# expect MATCH: the program's pthread_equal (a + b == 5; before: a comparison)
extown x_own_pthread_equal 0
# expect MATCH: the program's sched_get_priority_max/min (p + 100, p + 200; with the models attached:
# Linux's values, and -1 for 9, no policy)
extown x_own_sched_prio 1
extown x_own_sched_prio 9
# expect MATCH: the program's pthread_setschedparam (with the model: UB, 5 is no thread's id)
extown x_own_setschedparam 2
# expect MATCH: the program's strerror_r ('a' + e in buf; with the model: glibc's message)
extown x_own_strerror_r 3
# expect MATCH: the program's __errno_location (its own errno, 9 at start; with the model: the
# frontend's errno, 0 at start)
extown x_own_errno_location 4
# expect MATCH: a static function named like a C library function runs its body (internal linkage)
extown x_own_static_sqrt 0
